-- ============================================================
-- BASE DE DATOS: ZIMBRA_CRM
-- Objetivo: Gestionar contactos, prospectos, marketing y ventas.
-- ============================================================
 
CREATE DATABASE IF NOT EXISTS zimbra_crm CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE zimbra_crm;
 
-- ============================================================
-- 1. TABLA: ROLES
-- Almacena los perfiles de acceso de los usuarios.
-- Catalogo de roles y perfiles de acceso para los usuarios de Zimbra.
-- ============================================================
 
CREATE TABLE roles (
    id_rol SMALLINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(40) NOT NULL,
    descripcion VARCHAR(255),
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uq_roles_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 2. TABLA: USUARIOS
-- Almacena vendedores, administradores y personal de marketing.
-- Cuentas de acceso del personal de Zimbra que administra prospectos, clientes y ventas.
-- ============================================================
 
CREATE TABLE usuarios (
    id_usuario BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_rol SMALLINT UNSIGNED NOT NULL,
    nombre VARCHAR(80) NOT NULL,
    apellido VARCHAR(80) NOT NULL,
    correo VARCHAR(254) NOT NULL,
    nombre_usuario VARCHAR(50) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    telefono VARCHAR(30),
    activo TINYINT(1) NOT NULL DEFAULT 1,
    ultimo_acceso DATETIME,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_rol) REFERENCES roles(id_rol)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 3. TABLA: EMPRESAS
-- Almacena las organizaciones a las que pertenecen los contactos.
-- Empresas u organizaciones asociadas a los contactos comerciales de Zimbra.
-- ============================================================
 
CREATE TABLE empresas (
    id_empresa BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_legal VARCHAR(180) NOT NULL,
    nombre_comercial VARCHAR(180),
    nit VARCHAR(30),
    sitio_web VARCHAR(500),
    sector VARCHAR(100),
    direccion VARCHAR(255),
    ciudad VARCHAR(100),
    pais CHAR(2) NOT NULL DEFAULT 'CO',
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 4. TABLA: CONTACTOS
-- Almacena la información de las personas que interactúan
-- con Zimbra, sean prospectos o clientes.
-- Registro central de personas que interactúan con Zimbra; pueden ser prospectos y posteriormente clientes.
-- ============================================================
 
CREATE TABLE contactos (
    id_contacto BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_empresa BIGINT UNSIGNED,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    correo VARCHAR(254),
    telefono VARCHAR(30),
    cargo VARCHAR(120),
    consentimiento_email ENUM('PENDIENTE','OTORGADO','REVOCADO') NOT NULL DEFAULT 'PENDIENTE',
    fecha_consentimiento_email DATETIME,
    fecha_revocacion_email DATETIME,
    preferencia_contacto ENUM('EMAIL','TELEFONO','AMBOS','SIN_DEFINIR') NOT NULL DEFAULT 'SIN_DEFINIR',
    activo TINYINT(1) NOT NULL DEFAULT 1,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_empresa) REFERENCES empresas(id_empresa)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 5. TABLA: FUENTES_PROSPECTOS
-- Almacena los canales por los cuales se captan los prospectos.
-- Catalogo de canales de captación y origen de los prospectos.
-- ============================================================
 
CREATE TABLE fuentes_prospectos (
    id_fuente SMALLINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(80) NOT NULL,
    descripcion VARCHAR(255),
    activa TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 6. TABLA: PROSPECTOS
-- Administra los posibles clientes y su asignación a vendedores.
-- Gestiona el proceso de captación y calificación de prospectos y su asignación a vendedores.
-- ============================================================
 
CREATE TABLE prospectos (
    id_prospecto BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_contacto BIGINT UNSIGNED NOT NULL,
    id_fuente SMALLINT UNSIGNED NOT NULL,
    id_responsable BIGINT UNSIGNED,
    id_externo_crm VARCHAR(100),
    estado ENUM('NUEVO','CALIFICADO','EN_SEGUIMIENTO','CONVERTIDO','DESCARTADO') NOT NULL DEFAULT 'NUEVO',
    prioridad ENUM('BAJA','MEDIA','ALTA') NOT NULL DEFAULT 'MEDIA',
    puntuacion SMALLINT UNSIGNED NOT NULL DEFAULT 0,
    fecha_asignacion DATETIME,
    fecha_calificacion DATETIME,
    fecha_conversion DATETIME,
    motivo_descarte VARCHAR(500),
    observaciones TEXT,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_contacto) REFERENCES contactos(id_contacto),
    FOREIGN KEY (id_fuente) REFERENCES fuentes_prospectos(id_fuente),
    FOREIGN KEY (id_responsable) REFERENCES usuarios(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 7. TABLA: VISITANTES_WEB
-- Identifica a los visitantes del sitio web, incluso anónimos.
-- Identifica visitantes del sitio web y permite relacionarlos con contactos cuando son identificados.
-- ============================================================
 
CREATE TABLE visitantes_web (
    id_visitante BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    token_visitante CHAR(36) NOT NULL,
    id_contacto BIGINT UNSIGNED,
    consentimiento_tracking ENUM('DESCONOCIDO','ACEPTADO','RECHAZADO') NOT NULL DEFAULT 'DESCONOCIDO',
    primer_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ultimo_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_contacto) REFERENCES contactos(id_contacto)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 8. TABLA: EVENTOS_WEB
-- Registra las actividades realizadas por los visitantes.
-- Registra las visitas, formularios, clics y descargas realizadas en el sitio web de Zimbra.
-- ============================================================
 
CREATE TABLE eventos_web (
    id_evento_web BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_visitante BIGINT UNSIGNED NOT NULL,
    tipo_evento ENUM('VISITA_PAGINA','DESCARGA_PRUEBA','FORMULARIO','CLIC','OTRO') NOT NULL,
    url VARCHAR(2048),
    titulo_pagina VARCHAR(255),
    fecha_evento DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    metadatos JSON,
    FOREIGN KEY (id_visitante) REFERENCES visitantes_web(id_visitante)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 9. TABLA: PRODUCTOS
-- Almacena los productos y versiones comerciales de Zimbra.
-- Catalogo de productos y versiones comerciales ofrecidas por Zimbra.
-- ============================================================
 
CREATE TABLE productos (
    id_producto BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT,
    precio_lista DECIMAL(15,2),
    moneda CHAR(3) NOT NULL DEFAULT 'COP',
    activo TINYINT(1) NOT NULL DEFAULT 1,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT chk_productos_precio CHECK (precio_lista IS NULL OR precio_lista >= 0),
    UNIQUE KEY uq_productos_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 10. TABLA: PRUEBAS_PRODUCTO
-- Controla las pruebas gratuitas de los productos.
-- Controla las pruebas gratuitas por contacto y producto, incluyendo inicio, vencimiento y conversión.
-- ============================================================
 
CREATE TABLE pruebas_producto (
    id_prueba BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_contacto BIGINT UNSIGNED NOT NULL,
    id_producto BIGINT UNSIGNED NOT NULL,
    id_prospecto BIGINT UNSIGNED,
    fecha_inicio DATETIME NOT NULL,
    fecha_fin DATETIME NOT NULL,
    estado ENUM('ACTIVA','VENCIDA','FINALIZADA','CONVERTIDO','CANCELADA') NOT NULL DEFAULT 'ACTIVA',
    fecha_conversion DATETIME,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_contacto) REFERENCES contactos(id_contacto),
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto),
    FOREIGN KEY (id_prospecto) REFERENCES prospectos(id_prospecto),
    CONSTRAINT chk_pruebas_fechas CHECK (fecha_fin > fecha_inicio)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 11. TABLA: OPORTUNIDADES
-- Administra los posibles negocios hasta su cierre.
-- Gestiona las oportunidades comerciales, sus responsables, etapas y resultados de venta.
-- ============================================================
 
CREATE TABLE oportunidades (
    id_oportunidad BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_prospecto BIGINT UNSIGNED NOT NULL,
    id_responsable BIGINT UNSIGNED NOT NULL,
    id_externo_crm VARCHAR(100),
    nombre VARCHAR(180) NOT NULL,
    etapa ENUM('DESCUBRIMIENTO','CALIFICACION','PROPUESTA','NEGOCIACION','CIERRE') NOT NULL DEFAULT 'DESCUBRIMIENTO',
    estado ENUM('ABIERTA','GANADA','PERDIDA') NOT NULL DEFAULT 'ABIERTA',
    monto_estimado DECIMAL(15,2),
    moneda CHAR(3) NOT NULL DEFAULT 'COP',
    fecha_cierre_prevista DATE,
    fecha_cierre_real DATETIME,
    motivo_perdida VARCHAR(500),
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_prospecto) REFERENCES prospectos(id_prospecto),
    FOREIGN KEY (id_responsable) REFERENCES usuarios(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 12. TABLA: INTERACCIONES
-- Registra llamadas, correos, reuniones y seguimientos.
-- Registra las llamadas, correos, reuniones y actividades de seguimiento entre vendedores y contactos.
-- ============================================================
 
CREATE TABLE interacciones (
    id_interaccion BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_contacto BIGINT UNSIGNED NOT NULL,
    id_prospecto BIGINT UNSIGNED,
    id_oportunidad BIGINT UNSIGNED,
    id_usuario BIGINT UNSIGNED NOT NULL,
    tipo ENUM('LLAMADA','CORREO','REUNION','WHATSAPP','OTRO') NOT NULL,
    direccion ENUM('ENTRANTE','SALIENTE','NO_APLICA') NOT NULL DEFAULT 'SALIENTE',
    asunto VARCHAR(180),
    descripcion TEXT,
    resultado ENUM('CONTACTADO','SIN_RESPUESTA','INTERESADO','NO_INTERESADO','REPROGRAMAR','OTRO'),
    estado ENUM('PENDIENTE','COMPLETADA','CANCELADA') NOT NULL DEFAULT 'PENDIENTE',
    fecha_programada DATETIME,
    fecha_realizada DATETIME,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_contacto) REFERENCES contactos(id_contacto),
    FOREIGN KEY (id_prospecto) REFERENCES prospectos(id_prospecto),
    FOREIGN KEY (id_oportunidad) REFERENCES oportunidades(id_oportunidad),
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
 
-- ============================================================
-- 13. TABLA: ALERTAS
-- Almacena notificaciones para los vendedores y administradores.
-- Almacena las notificaciones destinadas a vendedores y administradores sobre actividades de prospectos.
-- ============================================================
 
CREATE TABLE alertas (
    id_alerta BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_usuario BIGINT UNSIGNED NOT NULL,
    id_prospecto BIGINT UNSIGNED NOT NULL,
    id_evento_web BIGINT UNSIGNED,
    id_interaccion BIGINT UNSIGNED,
    tipo_alerta ENUM('VISITA_RELEVANTE','DESCARGA_PRUEBA','SEGUIMIENTO','PRUEBA_POR_VENCER','OTRA') NOT NULL,
    titulo VARCHAR(180) NOT NULL,
    mensaje VARCHAR(1000),
    creada_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leida_en DATETIME,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (id_prospecto) REFERENCES prospectos(id_prospecto),
    FOREIGN KEY (id_evento_web) REFERENCES eventos_web(id_evento_web),
    FOREIGN KEY (id_interaccion) REFERENCES interacciones(id_interaccion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 14. TABLA: CAMPANAS
-- Administra las campañas de marketing por correo electrónico.
-- Almacena y administra las campañas de marketing por correo electrónico.
-- ============================================================
 
CREATE TABLE campanas (
    id_campana BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_creador BIGINT UNSIGNED NOT NULL,
    nombre VARCHAR(180) NOT NULL,
    asunto_correo VARCHAR(255) NOT NULL,
    contenido MEDIUMTEXT NOT NULL,
    estado ENUM('BORRADOR','PROGRAMADA','ENVIADA','CANCELADA') NOT NULL DEFAULT 'BORRADOR',
    programada_para DATETIME,
    enviada_en DATETIME,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_creador) REFERENCES usuarios(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- ============================================================
-- 15. TABLA: ENVIOS_CAMPANA
-- Registra los correos enviados a cada contacto en una campaña.
-- Registra los envíos individuales de campañas y sus resultados de entrega, apertura y clics.
-- ============================================================
 
CREATE TABLE envios_campana (
    id_envio BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_campana BIGINT UNSIGNED NOT NULL,
    id_contacto BIGINT UNSIGNED NOT NULL,
    estado ENUM('PENDIENTE','ENVIADO','ENTREGADO','ABIERTO','CLIC','REBOTADO','CANCELADO') NOT NULL DEFAULT 'PENDIENTE',
    enviado_en DATETIME,
    abierto_en DATETIME,
    clic_en DATETIME,
    detalle_error VARCHAR(500),
    FOREIGN KEY (id_campana) REFERENCES campanas(id_campana),
    FOREIGN KEY (id_contacto) REFERENCES contactos(id_contacto),
    UNIQUE KEY uq_envios_campana_contacto (id_campana, id_contacto)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
 
 
-- Indices unicos para evitar duplicados incluso ante solicitudes simultaneas.
ALTER TABLE usuarios
  ADD UNIQUE KEY uq_usuarios_correo (correo),
  ADD UNIQUE KEY uq_usuarios_nombre_usuario (nombre_usuario);
 
ALTER TABLE visitantes_web
  ADD UNIQUE KEY uq_visitantes_token (token_visitante);
 
-- Bitacora requerida por los triggers y procedimientos almacenados.
CREATE TABLE bitacora_uso (
    id_bitacora BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    id_usuario BIGINT UNSIGNED NULL,
    tipo_evento VARCHAR(60) NOT NULL,
    detalle TEXT,
    fecha_evento DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
      ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
 
-- Indices de consulta complementarios para las relaciones y filtros del CRM.
ALTER TABLE empresas ADD UNIQUE KEY uq_empresas_nit (nit);
ALTER TABLE fuentes_prospectos ADD UNIQUE KEY uq_fuentes_nombre (nombre);
ALTER TABLE contactos ADD KEY idx_contactos_empresa_activo (id_empresa, activo);
ALTER TABLE prospectos ADD KEY idx_prospectos_estado_prioridad (estado, prioridad);
ALTER TABLE prospectos ADD KEY idx_prospectos_responsable_estado (id_responsable, estado);
ALTER TABLE visitantes_web ADD KEY idx_visitantes_contacto (id_contacto);
ALTER TABLE eventos_web ADD KEY idx_eventos_visitante_fecha (id_visitante, fecha_evento);
ALTER TABLE pruebas_producto ADD KEY idx_pruebas_estado_vencimiento (estado, fecha_fin);
ALTER TABLE oportunidades ADD KEY idx_oportunidades_estado_etapa (estado, etapa);
ALTER TABLE interacciones ADD KEY idx_interacciones_prospecto_fecha (id_prospecto, fecha_realizada);
ALTER TABLE alertas ADD KEY idx_alertas_usuario_lectura_fecha (id_usuario, leida_en, creada_en);
ALTER TABLE envios_campana ADD KEY idx_envios_contacto_estado (id_contacto, estado);
ALTER TABLE bitacora_uso ADD KEY idx_bitacora_usuario_fecha (id_usuario, fecha_evento);
 
USE zimbra_crm;
DELIMITER //
 
-- T1. Antes de crear un usuario, valida el rol y los datos minimos.
DROP TRIGGER IF EXISTS trg_validar_usuario_insercion //
CREATE TRIGGER trg_validar_usuario_insercion
BEFORE INSERT ON usuarios FOR EACH ROW
BEGIN
  IF NOT EXISTS (SELECT 1 FROM roles WHERE id_rol = NEW.id_rol) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El rol no existe';
  END IF;
  IF NEW.correo IS NULL OR NEW.correo NOT LIKE '%_@_%.__%' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Correo con formato invalido';
  END IF;
  -- Los indices UNIQUE de la tabla impiden duplicados incluso en altas simultaneas.
END //
 
-- T2. Mantiene las mismas validaciones cuando se modifica un usuario.
DROP TRIGGER IF EXISTS trg_validar_usuario_actualizacion //
CREATE TRIGGER trg_validar_usuario_actualizacion
BEFORE UPDATE ON usuarios FOR EACH ROW
BEGIN
  IF NOT (OLD.id_rol <=> NEW.id_rol)
     AND NOT EXISTS (SELECT 1 FROM roles WHERE id_rol = NEW.id_rol) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El nuevo rol no existe';
  END IF;
  IF NOT (OLD.correo <=> NEW.correo) THEN
    IF NEW.correo IS NULL OR NEW.correo NOT LIKE '%_@_%.__%' THEN
      SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Correo con formato invalido';
    END IF;
    -- El indice UNIQUE uq_usuarios_correo valida que no pertenezca a otra cuenta.
  END IF;
  -- El indice UNIQUE uq_usuarios_nombre_usuario evita nombres repetidos.
END //
 
-- T3. Registra en bitacora cada usuario nuevo.
DROP TRIGGER IF EXISTS trg_registrar_creacion_usuario //
CREATE TRIGGER trg_registrar_creacion_usuario
AFTER INSERT ON usuarios FOR EACH ROW
BEGIN
  INSERT INTO bitacora_uso(id_usuario, tipo_evento, detalle)
  VALUES (NEW.id_usuario, 'CREACION_USUARIO',
          CONCAT('Usuario ', NEW.nombre_usuario, ' creado con rol ', NEW.id_rol));
END //
 
-- T4. Audita acceso, cambio de rol y activacion/desactivacion.
DROP TRIGGER IF EXISTS trg_auditar_usuario_actualizacion //
CREATE TRIGGER trg_auditar_usuario_actualizacion
AFTER UPDATE ON usuarios FOR EACH ROW
BEGIN
  IF NOT (OLD.ultimo_acceso <=> NEW.ultimo_acceso) AND NEW.ultimo_acceso IS NOT NULL THEN
    INSERT INTO bitacora_uso(id_usuario, tipo_evento, detalle)
    VALUES (NEW.id_usuario, 'ACCESO_SISTEMA', CONCAT('Acceso registrado: ', NEW.ultimo_acceso));
  END IF;
  IF NOT (OLD.id_rol <=> NEW.id_rol) THEN
    INSERT INTO bitacora_uso(id_usuario, tipo_evento, detalle)
    VALUES (NEW.id_usuario, 'CAMBIO_ROL', CONCAT('De rol ', OLD.id_rol, ' a ', NEW.id_rol));
  END IF;
  IF NOT (OLD.activo <=> NEW.activo) THEN
    INSERT INTO bitacora_uso(id_usuario, tipo_evento, detalle)
    VALUES (NEW.id_usuario, IF(NEW.activo=1,'USUARIO_ACTIVADO','USUARIO_DESACTIVADO'), 'Cambio de estado');
  END IF;
END //
 
-- T5. Solo permite agregar envios si la campana esta programada/enviada y hay consentimiento.
DROP TRIGGER IF EXISTS trg_validar_envio_campana //
CREATE TRIGGER trg_validar_envio_campana
BEFORE INSERT ON envios_campana FOR EACH ROW
BEGIN
  DECLARE v_consentimiento VARCHAR(20) DEFAULT NULL;
  DECLARE v_estado_camp VARCHAR(20) DEFAULT NULL;
  SELECT consentimiento_email INTO v_consentimiento
    FROM contactos WHERE id_contacto = NEW.id_contacto;
  IF v_consentimiento IS NULL OR v_consentimiento <> 'OTORGADO' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El contacto no otorgo consentimiento para correo';
  END IF;
  SELECT estado INTO v_estado_camp FROM campanas WHERE id_campana = NEW.id_campana;
  IF v_estado_camp IS NULL OR v_estado_camp NOT IN ('PROGRAMADA','ENVIADA') THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La campana no esta en estado valido para enviar';
  END IF;
END //
 
-- T6. Genera una alerta cuando una interaccion requiere seguimiento comercial.
DROP TRIGGER IF EXISTS trg_generar_alerta_seguimiento //
CREATE TRIGGER trg_generar_alerta_seguimiento
AFTER INSERT ON interacciones FOR EACH ROW
BEGIN
  IF NEW.id_prospecto IS NOT NULL AND NEW.resultado IN ('INTERESADO','REPROGRAMAR') THEN
    INSERT INTO alertas(id_usuario, id_prospecto, id_interaccion, tipo_alerta, titulo, mensaje)
    VALUES (NEW.id_usuario, NEW.id_prospecto, NEW.id_interaccion, 'SEGUIMIENTO',
            'Seguimiento pendiente',
            CONCAT('Interaccion con resultado ', NEW.resultado, '. Realizar seguimiento.'));
  END IF;
END //
 
-- T7. Deja evidencia de los cambios de estado de los prospectos.
DROP TRIGGER IF EXISTS trg_registrar_cambio_estado_prospecto //
CREATE TRIGGER trg_registrar_cambio_estado_prospecto
AFTER UPDATE ON prospectos FOR EACH ROW
BEGIN
  IF NOT (OLD.estado <=> NEW.estado) THEN
    INSERT INTO bitacora_uso(id_usuario, tipo_evento, detalle)
    VALUES (NEW.id_responsable, 'CAMBIO_ESTADO_PROSPECTO',
            CONCAT('Prospecto ', NEW.id_prospecto, ': ', OLD.estado, ' -> ', NEW.estado));
  END IF;
END //
 
-- T8. Impide rangos de prueba incoherentes y completa la fecha de conversion.
DROP TRIGGER IF EXISTS trg_validar_prueba_producto_insert //
CREATE TRIGGER trg_validar_prueba_producto_insert
BEFORE INSERT ON pruebas_producto FOR EACH ROW
BEGIN
  IF NEW.fecha_fin <= NEW.fecha_inicio THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La fecha de fin debe ser posterior al inicio';
  END IF;
  IF NEW.id_prospecto IS NOT NULL AND NOT EXISTS (
      SELECT 1 FROM prospectos p
      WHERE p.id_prospecto = NEW.id_prospecto AND p.id_contacto = NEW.id_contacto) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El prospecto no corresponde al contacto de la prueba';
  END IF;
  IF NEW.estado = 'CONVERTIDO' AND NEW.fecha_conversion IS NULL THEN
    SET NEW.fecha_conversion = NOW();
  END IF;
END //

-- Demostración ejecutable desde MySQL y Django. El SAVEPOINT es temporal
-- dentro de la transacción; la bitácora conserva evidencia del rollback.
DROP PROCEDURE IF EXISTS sp_demo_savepoint_auditable //
CREATE PROCEDURE sp_demo_savepoint_auditable(
  IN p_id_usuario BIGINT UNSIGNED,
  OUT p_estado VARCHAR(10),
  OUT p_mensaje VARCHAR(200)
)
BEGIN
  DECLARE v_id_temporal BIGINT UNSIGNED DEFAULT NULL;
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    ROLLBACK;
    SET p_estado='ERROR';
    SET p_mensaje='No fue posible ejecutar la demostración SAVEPOINT';
  END;
 
  START TRANSACTION;
  IF NOT EXISTS (SELECT 1 FROM usuarios WHERE id_usuario=p_id_usuario AND activo=1) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Usuario inexistente o inactivo';
  END IF;
 
  INSERT INTO bitacora_uso(id_usuario,tipo_evento,detalle)
  VALUES(p_id_usuario,'SAVEPOINT_INICIO',
         'Demostración iniciada: se crea un punto de restauración dentro de una transacción.');
 
  SAVEPOINT sp_demo_restauracion;
 
  -- Este registro es deliberadamente temporal y debe desaparecer al revertir.
  INSERT INTO bitacora_uso(id_usuario,tipo_evento,detalle)
  VALUES(p_id_usuario,'SAVEPOINT_TEMPORAL',
         'Registro temporal de prueba: debe eliminarse con ROLLBACK TO SAVEPOINT.');
  SET v_id_temporal=LAST_INSERT_ID();
 
  ROLLBACK TO SAVEPOINT sp_demo_restauracion;
  RELEASE SAVEPOINT sp_demo_restauracion;
 
  -- Esta evidencia se escribe después del rollback y sí queda confirmada.
  INSERT INTO bitacora_uso(id_usuario,tipo_evento,detalle)
  VALUES(p_id_usuario,'SAVEPOINT_ROLLBACK_OK',
         CONCAT('Rollback parcial ejecutado correctamente. El registro temporal ',
                v_id_temporal,' fue revertido; la transacción continuó y se confirmó.'));
 
  COMMIT;
  SET p_estado='OK';
  SET p_mensaje='SAVEPOINT ejecutado: la operación temporal se revirtió y la evidencia quedó en bitacora_uso.';
END //
